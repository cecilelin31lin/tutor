<script>

const pairs = [

{ image: 'https://raw.githubusercontent.com/cecilelin31lin/pic/main/spring.png', sentence: 'spring' },

{ image: 'https://raw.githubusercontent.com/cecilelin31lin/pic/main/fall.png', sentence: 'fall' },

{ image: 'https://raw.githubusercontent.com/cecilelin31lin/pic/main/summer.png', sentence: 'summer' },

{ image: 'https://raw.githubusercontent.com/cecilelin31lin/pic/main/winter.png', sentence: 'winter' },

];


let selectedImageIdx = null;

let matchedCount = 0;

let matchedImages = [];

let startTime = Date.now();

let timerInterval = null;


function initGame() {

const shuffled = [...pairs].sort(() => Math.random() - 0.5);


// 更新配對計數顯示

document.getElementById('matched').textContent = '0/' + pairs.length;


// 生成圖片按鈕

const imageButtonsContainer = document.getElementById('imageButtonsContainer');

pairs.forEach((item, idx) => {

const btn = document.createElement('button');

btn.className = 'image-btn';

btn.dataset.index = idx;

btn.innerHTML = '<img src="' + item.image + '" alt="image">';

btn.onclick = () => selectImage(btn, idx);

imageButtonsContainer.appendChild(btn);

});


// 生成句子按鈕

const sentencesContainer = document.getElementById('sentencesContainer');

shuffled.forEach((item, idx) => {

const btn = document.createElement('button');

btn.className = 'sentence-btn';

btn.textContent = item.sentence;

btn.setAttribute('data-image-idx', pairs.findIndex(p => p.image === item.image));

btn.onclick = () => selectSentence(btn);

sentencesContainer.appendChild(btn);

});


startTimer();

}


function selectImage(btn, idx) {

const prevSelected = document.querySelector('.image-btn[data-selected="true"]');

if (prevSelected) {

prevSelected.removeAttribute('data-selected');

}


selectedImageIdx = idx;

btn.setAttribute('data-selected', 'true');

}


function selectSentence(btn) {

if (selectedImageIdx === null) return;


const correctImageIdx = parseInt(btn.getAttribute('data-image-idx'));


if (selectedImageIdx === correctImageIdx) {

btn.classList.add('correct');

btn.onclick = null;


matchedImages.push(pairs[selectedImageIdx].image);

matchedCount++;

document.getElementById('matched').textContent = matchedCount + '/' + pairs.length;


const selectedBtn = document.querySelector('.image-btn[data-selected="true"]');

if (selectedBtn) {

selectedBtn.style.opacity = '0.5';

selectedBtn.onclick = null;

}


selectedImageIdx = null;


if (matchedCount === pairs.length) {

showResults();

}

} else {

btn.classList.add('incorrect');

setTimeout(() => {

btn.classList.remove('incorrect');

}, 500);

}

}


function startTimer() {

startTime = Date.now();

timerInterval = setInterval(() => {

const elapsed = Math.floor((Date.now() - startTime) / 1000);

const mins = Math.floor(elapsed / 60);

const secs = elapsed % 60;

document.getElementById('timer').textContent = mins + ':' + (secs < 10 ? '0' : '') + secs;

}, 100);

}


function showResults() {

clearInterval(timerInterval);


const elapsed = Math.floor((Date.now() - startTime) / 1000);

const mins = Math.floor(elapsed / 60);

const secs = elapsed % 60;

const timeStr = mins + ':' + (secs < 10 ? '0' : '') + secs;


document.getElementById('finalTime').textContent = timeStr;


const imagesHTML = matchedImages.map(image =>

`<div class="image-tag"><img src="${image}" alt="image"></div>`

).join('');

document.getElementById('finalImages').innerHTML = imagesHTML;


document.getElementById('resultOverlay').classList.add('show');

}


initGame();

</script>
